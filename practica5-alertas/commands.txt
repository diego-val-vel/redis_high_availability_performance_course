-- Modelo de datos mínimo para alertas

/*
	Convención de claves:

	alerts:seq STRING contador global de alertas
	alerts:{type}:list LIST últimas N alertas por tipo
	alerts:{type}:last HASH última alerta por tipo (para lectura rápida)
	alerts:recent LIST ring buffer global recortado
	alerts:pub canal Pub/Sub para notificaciones en tiempo real
*/

-- Inserts seguros con TTL y recorte

# Crear o inicializar contador global de alertas si no existe
docker compose exec -T redis redis-cli SETNX "alerts:seq" 0

# Publicar y guardar una alerta de tipo "payment" con TTL en la vista "last"
# 1) Incrementa el id global (atómico)
docker compose exec -T redis redis-cli INCR "alerts:seq"

# 2) Prepara un objeto simple en HASH como "última alerta" para lectura inmediata
docker compose exec -T redis redis-cli HSET "alerts:{payment}:last" id "1" level "warn" msg "saldo bajo" ts "1731360000"

# 3) Opcional: expiración de la "vista last" para no perpetuar info vieja (ej 10 min)
docker compose exec -T redis redis-cli EXPIRE "alerts:{payment}:last" 600

# 4) Guardar la alerta en la lista "por tipo", recientes primero
docker compose exec -T redis redis-cli LPUSH "alerts:{payment}:list" "1|warn|saldo bajo|1731360000"

# 5) Mantener solo las últimas 100 por tipo
docker compose exec -T redis redis-cli LTRIM "alerts:{payment}:list" 0 99

# 6) Guardar también en la lista global "recent" para dashboards
docker compose exec -T redis redis-cli LPUSH "alerts:recent" "payment|1|warn|saldo bajo|1731360000"
docker compose exec -T redis redis-cli LTRIM "alerts:recent" 0 499

# 7) Notificar en tiempo real por Pub/Sub
docker compose exec -T redis redis-cli PUBLISH "alerts:pub" "payment|1|warn|saldo bajo|1731360000"

-- Lecturas rápidas (SELECTs)

# Obtener la última alerta de "payment" como pares campo/valor (lectura puntual)
docker compose exec -T redis redis-cli HGETALL "alerts:{payment}:last"

# Obtener top 5 recientes por tipo
docker compose exec -T redis redis-cli LRANGE "alerts:{payment}:list" 0 4

# Obtener top 5 globales
docker compose exec -T redis redis-cli LRANGE "alerts:recent" 0 4

-- Pub/Sub en dos terminales
-- Terminal 1 suscriptor:

# Suscribirse al canal de alertas en tiempo real
docker compose exec -T redis redis-cli SUBSCRIBE "alerts:pub"

-- Terminal 2 publicador:

# Publicar una nueva alerta (recibirás el mensaje en la otra terminal)
docker compose exec -T redis redis-cli PUBLISH "alerts:pub" "refund|2|info|reembolso iniciado|1731360060"

-- Transacciones y Lua para atomicidad
-- A) WATCH + MULTI + EXEC para contador por tipo

# Queremos incrementar un contador por tipo de alerta solo si la secuencia global no cambió.
# 1) Vigilar la clave de secuencia
docker compose exec -T redis redis-cli WATCH "alerts:seq"

# 2) Leer el valor actual de seq (simula lógica de app)
docker compose exec -T redis redis-cli GET "alerts:seq"

# 3) Inicia transacción
docker compose exec -T redis redis-cli MULTI

# 4) Incremento por tipo "payment" (pendiente hasta EXEC)
docker compose exec -T redis redis-cli INCR "alerts:{payment}:count"

# 5) Ejecuta transacción (si "alerts:seq" fue modificada por otros entre WATCH y EXEC, fallará)
docker compose exec -T redis redis-cli EXEC

-- B) Script Lua que guarda y publica en un solo paso

# Script que: LPUSH en recent, LTRIM, y PUBLISH un mensaje. Todo atómico.
docker compose exec -T redis redis-cli EVAL '
  local msg = ARGV[1]
  redis.call("LPUSH", "alerts:recent", msg)
  redis.call("LTRIM", "alerts:recent", 0, 499)
  redis.call("PUBLISH", "alerts:pub", msg)
  return "OK"
' 0 "inventory|3|crit|stock agotado|1731360120"

-- Qué hacer si tu cliente no trae un comando
-- A) Primero valida en redis-cli

# Ver documentación de un comando (ayuda rápida)

# Invocar un CLIENT INFO via Lua (ejemplo de wrapper cuando el cliente no expone un comando)
docker compose exec -T redis redis-cli EVAL 'return redis.call("CLIENT","INFO")' 0

/*
	C) Alternativas

	Actualiza el SDK a versión que soporte el comando.

	Abre issue en el repositorio del cliente.

	Usa un comando alterno o un script Lua que encapsule la semántica.

	Por qué algunos comandos no aparecen en clientes

	Seguridad: comandos como DEBUG, FLUSHALL pueden estar escondidos por diseño.

	Compatibilidad: el cliente no alcanzó la versión del server.

	Política del proveedor: eliminan APIs que no consideran “safe”.
	Cómo proceder:

	Verifica en redis-cli que el comando existe y su versión.

	Si es seguro, usa Lua wrapper temporal.

	Evita comandos bloqueantes o peligrosos en producción.
*/
