CHULETA MODULO 6 - LABORATORIO REDIS HA
======================================

# 0. ARRANQUE DEL LABORATORIO
# ===========================

# 0.1 Dar permisos al script
chmod +x start_mod6.sh

# 0.2 Levantar todo el entorno:
#    - redis-master, redis-replica-1, redis-extra, redis-insight (con docker-compose.ha.yml)
#    - Y LUEGO los 3 sentinels (redis-sentinel-1..3) usando la IP real de redis-master
./start_mod6.sh


# 1. VERIFICAR CONTENEDORES
# =========================

docker ps
# Debes ver al menos:
# - redis-master
# - redis-replica-1
# - redis-extra
# - redis-sentinel-1
# - redis-sentinel-2
# - redis-sentinel-3
# - redis-insight


# 2. VER ROLES MASTER / REPLICA / INSTANCIA EXTRA
# ===============================================

docker exec -it redis-master redis-cli INFO replication
# Esperado:
#   role:master
#   connected_slaves:1
#   slave0: ... redis-replica-1 ...

docker exec -it redis-replica-1 redis-cli INFO replication
# Esperado:
#   role:slave (o role:replica)
#   master_host:redis-master  (o la IP interna del master)
#   master_link_status:up

docker exec -it redis-extra redis-cli INFO server
# Confirma que redis-extra es una instancia standalone separada
# (redis_mode:standalone, tcp_port:6379, etc).


# 3. PROBAR REPLICACION
# =====================

docker exec -it redis-master redis-cli SET usuario:1:nombre "Diego"
# Escritura en el master

docker exec -it redis-replica-1 redis-cli GET usuario:1:nombre
# Lectura en la replica: debe devolver "Diego" si la replicacion funciona.


# 4. VER ESTADO DE SENTINEL
# =========================
# Estos comandos funcionan porque los sentinels
# se crearon desde start_mod6.sh con la IP correcta del master.

docker exec -it redis-sentinel-1 redis-cli -p 26379 INFO sentinel
# Esperado:
#   sentinel_masters:1
#   master0:name=mymaster,status=ok,...
#   slaves=1 (redis-replica-1)
#   sentinels=3

docker exec -it redis-sentinel-1 redis-cli -p 26379 SENTINEL masters
# Lista los masters que monitorea (debe aparecer "mymaster" con ip/puerto del master)

docker exec -it redis-sentinel-1 redis-cli -p 26379 SENTINEL master mymaster
# Detalle del master:
#   "ip"            -> IP interna redis-master
#   "port"          -> 6379
#   "num-slaves"    -> 1
#   "num-other-sentinels" -> 2
#   "quorum"        -> 2


# 5. SIMULAR CAIDA DEL MASTER Y OBSERVAR FAILOVER
# ===============================================

docker stop redis-master
# Simula caida del master original

docker logs -f redis-sentinel-1
# Ver logs en tiempo real (Ctrl + C para salir).
# Deberias ver algo como:
#  - +sdown / +odown sobre mymaster
#  - +failover-state-select-slave
#  - +selected-slave ...
#  - +promoted-slave ...
#  - +switch-master mymaster <vieja-ip> 6379 <nueva-ip> 6379


# 6. CONFIRMAR NUEVO MASTER
# =========================

docker exec -it redis-replica-1 redis-cli INFO replication
# AHORA deberia mostrar:
#   role:master      (fue promovido)
#   connected_slaves:0 (por ahora, no le colgamos replicas)
#   master_host ya no aplica porque es master

docker exec -it redis-replica-1 redis-cli GET usuario:1:nombre
# La clave "usuario:1:nombre" debe seguir existiendo ("Diego")

# Probar nuevas escrituras en el nuevo master (antes era replica read-only)
docker exec -it redis-replica-1 redis-cli SET usuario:2:nombre "Ana"
docker exec -it redis-replica-1 redis-cli SET usuario:4:nombre "Luis"
# NO debe salir READONLY: ahora acepta escrituras.


# 7. SHARDING LOGICO (SHARD A Y SHARD B)
# ======================================

# Shard A = redis-replica-1 (nuevo master despues del failover)
# Shard B = redis-extra

# Usuarios pares al Shard A
docker exec -it redis-replica-1 redis-cli SET usuario:2:nombre "Ana"
docker exec -it redis-replica-1 redis-cli SET usuario:4:nombre "Luis"

# Usuarios impares al Shard B
docker exec -it redis-extra redis-cli SET usuario:1:nombre "Diego"
docker exec -it redis-extra redis-cli SET usuario:3:nombre "Maria"

# Ver claves en cada shard
docker exec -it redis-replica-1 redis-cli KEYS "usuario:*:nombre"
# Debe listar solo los usuarios que guardaste en Shard A

docker exec -it redis-extra redis-cli KEYS "usuario:*:nombre"
# Debe listar solo los usuarios que guardaste en Shard B


# 8. BACKUP Y RESTORE
# ===================

# 8.1 Forzar un snapshot RDB en el master actual (redis-replica-1)
docker exec -it redis-replica-1 redis-cli BGSAVE

# 8.2 Copiar el dump al host
docker cp redis-replica-1:/data/dump.rdb ./backup-dump.rdb

# 8.3 Preparar carpeta para restauracion
mkdir -p restore-data

# 8.4 Copiar el dump a la carpeta de restauracion con el nombre que espera Redis
cp ./backup-dump.rdb ./restore-data/dump.rdb

# 8.5 Levantar un nuevo contenedor que arranque desde el dump
docker run -d --name redis-restore -p 6390:6379 -v "$(pwd)/restore-data:/data" redis:7-alpine

# 8.6 Ver claves restauradas
docker exec -it redis-restore redis-cli KEYS "usuario:*:nombre"
# Debes ver las claves que existian al momento de BGSAVE.


# 9. PARAMETROS DE CONFIGURACION Y SLOWLOG
# ========================================

docker exec -it redis-replica-1 redis-cli CONFIG GET appendonly
docker exec -it redis-replica-1 redis-cli CONFIG GET appendfsync
docker exec -it redis-replica-1 redis-cli CONFIG GET maxmemory

docker exec -it redis-replica-1 redis-cli SLOWLOG GET 10
# Muestra las ultimas 10 operaciones lentas (si hay).


# 10. LIMPIEZA DEL ENTORNO
# ========================

# 10.1 Apagar y borrar stack principal (master, replica, extra, insight)
docker compose -f docker-compose.ha.yml down -v

# 10.2 Apagar y borrar los sentinels (NO son parte de docker-compose)
docker rm -f redis-sentinel-1 redis-sentinel-2 redis-sentinel-3

# 10.3 Apagar y borrar el contenedor de restauracion
docker stop redis-restore
docker rm redis-restore

# 10.4 Borrar archivos de backup y restauracion (opcional)
rm -f backup-dump.rdb
rm -rf restore-data
