########################################
# 0. LIMPIEZA INICIAL (OPCIONAL)
########################################

# Verifica que estás en el directorio donde está docker-compose.ha.yml
pwd

# Si quieres asegurarte de que no quede nada de ejecuciones anteriores:
docker compose -f docker-compose.ha.yml down -v


########################################
# 1. LEVANTAR EL LABORATORIO HA (MÓDULO 6)
########################################

# Levantar todos los servicios definidos en docker-compose.ha.yml en segundo plano
docker compose -f docker-compose.ha.yml up -d

# Verificar que todos los contenedores estén arriba
docker ps


########################################
# 2. VER ROLES DE LAS INSTANCIAS REDIS
########################################

# Ver información de replicación en el nodo master
docker exec -it redis-master redis-cli INFO replication

# Ver información de replicación en la replica
docker exec -it redis-replica-1 redis-cli INFO replication

# Ver información básica del servidor en redis-extra (instancia independiente)
docker exec -it redis-extra redis-cli INFO server


########################################
# 3. PROBAR LA REPLICACIÓN MASTER / REPLICA
########################################

# Escribir una clave en el master
docker exec -it redis-master redis-cli SET usuario:1:nombre "Diego"

# Leer la misma clave desde la replica para comprobar replicación
docker exec -it redis-replica-1 redis-cli GET usuario:1:nombre


########################################
# 4. VER ESTADO DE SENTINEL
########################################

# Consultar información de Sentinel en el primer nodo
docker exec -it redis-sentinel-1 redis-cli -p 26379 INFO sentinel

# (Opcional) Ver maestros que está monitoreando Sentinel
docker exec -it redis-sentinel-1 redis-cli -p 26379 SENTINEL masters

# (Opcional) Ver detalle del master "mymaster" desde Sentinel
docker exec -it redis-sentinel-1 redis-cli -p 26379 SENTINEL master mymaster


########################################
# 5. SIMULAR FALLA DEL MASTER Y VER FAILOVER
########################################

# Detener el contenedor del master para simular una caída
docker stop redis-master

# Ver logs en tiempo real de un Sentinel para observar el failover (Ctrl + C para salir)
docker logs -f redis-sentinel-1

# Ver el rol actual de redis-replica-1 tras el failover (debería ser master)
docker exec -it redis-replica-1 redis-cli INFO replication

# Verificar que la clave original sigue existiendo en el nuevo master
docker exec -it redis-replica-1 redis-cli GET usuario:1:nombre


########################################
# 6. SHARDING LÓGICO (SHARD A = MASTER ACTUAL, SHARD B = redis-extra)
########################################

# Escribir usuarios "pares" en Shard A (master actual, redis-replica-1 tras failover)
docker exec -it redis-replica-1 redis-cli SET usuario:2:nombre "Ana"
docker exec -it redis-replica-1 redis-cli SET usuario:4:nombre "Luis"

# Escribir usuarios "impares" en Shard B (redis-extra)
docker exec -it redis-extra redis-cli SET usuario:1:nombre "Diego"
docker exec -it redis-extra redis-cli SET usuario:3:nombre "María"

# Ver claves de usuarios en Shard A
docker exec -it redis-replica-1 redis-cli KEYS "usuario:*:nombre"

# Ver claves de usuarios en Shard B
docker exec -it redis-extra redis-cli KEYS "usuario:*:nombre"


########################################
# 7. MONITOREO CON REDISINSIGHT
########################################

# Verificar que el contenedor de RedisInsight está corriendo
docker ps | grep redis-insight

# Luego, en el navegador, abrir:
# http://localhost:5540


########################################
# 8. BACKUP Y RESTORE BÁSICO
########################################

# Forzar un snapshot RDB en el master actual (redis-replica-1 si ya es master)
docker exec -it redis-replica-1 redis-cli BGSAVE

# Copiar el archivo dump.rdb desde el contenedor al host
docker cp redis-replica-1:/data/dump.rdb ./backup-dump.rdb

# Crear un directorio local para la restauración
mkdir -p restore-data

# Copiar el backup al directorio de restauración con el nombre que espera Redis
cp ./backup-dump.rdb ./restore-data/dump.rdb

# Levantar un nuevo contenedor Redis que cargue el dump de restore-data
docker run -d --name redis-restore -p 6390:6379 -v "$(pwd)/restore-data:/data" redis:7-alpine

# Ver las claves restauradas en el nuevo contenedor
docker exec -it redis-restore redis-cli KEYS "usuario:*:nombre"


########################################
# 9. TUNING / REVISIÓN DE CONFIGURACIÓN Y SLOWLOG
########################################

# Ver si el AOF está habilitado
docker exec -it redis-replica-1 redis-cli CONFIG GET appendonly

# Ver política de fsync para el AOF
docker exec -it redis-replica-1 redis-cli CONFIG GET appendfsync

# Ver límite de memoria configurado
docker exec -it redis-replica-1 redis-cli CONFIG GET maxmemory

# Ver las últimas 10 entradas del slowlog
docker exec -it redis-replica-1 redis-cli SLOWLOG GET 10


########################################
# 10. LIMPIEZA DEL LABORATORIO
########################################

# Bajar y borrar contenedores + volúmenes del stack principal
docker compose -f docker-compose.ha.yml down -v

# Detener el contenedor de restauración
docker stop redis-restore

# Eliminar el contenedor de restauración
docker rm redis-restore

# Borrar archivo de backup
rm -f backup-dump.rdb

# Borrar el directorio usado para la restauración
rm -rf restore-data
